    ensure(0<=offset<=image.image_size and 0<=length<=image.image_size-offset,'Image read outside logical bounds')
    image.seek(offset);data=image.read(length);ensure(len(data)==length,'Short image read');return data

def parse_avb(name):
    image=images[name]
    f,h,descs,_=avb.Avb()._parse_image(image)
    size=h.SIZE+h.authentication_data_block_size+h.auxiliary_data_block_size
    blob=readat(image,f.vbmeta_offset if f else 0,size)
    ensure(avb.verify_vbmeta_signature(h,blob),'Invalid vbmeta self-signature: '+name)
    if f:
        ensure(0<f.original_image_size<=f.vbmeta_offset and f.vbmeta_offset+f.vbmeta_size<=image.image_size-64,'Invalid footer bounds: '+name)
    result={'algorithm':avb.lookup_algorithm_by_type(h.algorithm_type)[0],'flags':h.flags,
        'rollback_index':h.rollback_index,'rollback_index_location':h.rollback_index_location,
        'signed':h.algorithm_type!=0,'vbmeta_blob_sha256':hashlib.sha256(blob).hexdigest(),
        'original_image_size':f.original_image_size if f else None,'descriptors':[]}
    for d in descs:
        item={'type':type(d).__name__,'encoded_sha256':hashlib.sha256(d.encode()).hexdigest()}
        for field in ['partition_name','image_size','hash_algorithm','flags','tree_offset','tree_size','data_block_size','hash_block_size','fec_num_roots','fec_offset','fec_size','rollback_index_location','key','value']:
            if hasattr(d,field):
                v=getattr(d,field);item[field]=v.decode(errors='replace') if isinstance(v,bytes) else v
        for field in ['salt','digest','root_digest','public_key']:
            if hasattr(d,field):item[field]=getattr(d,field).hex()
        result['descriptors'].append(item)
    meta[name]=(f,h,descs)
    return result

def require_embedded_descriptor(name, descriptor):
    # avbtool supports persistent digests: an empty digest/root is valid format
    # but deliberately skips the payload comparison. This offline audit has no
    # device persistent store and requires a supported embedded digest layout.
    if name in ('boot', 'dtbo'):
        expected_type = avb.AvbHashDescriptor
        field = 'digest'
        algorithms = ('sha256',)
    else:
        ensure(name in PARTITIONS, 'Unsupported AVB partition: '+name)
        expected_type = avb.AvbHashtreeDescriptor
        field = 'root_digest'
        # AOSP add_hashtree_footer still defaults to SHA1; SHA256 is its
        # supported nonpersistent alternative. Record the actual selection.
        algorithms = ('sha1', 'sha256')
    ensure(type(descriptor) is expected_type, 'Wrong AVB descriptor type: '+name)
    ensure(descriptor.hash_algorithm in algorithms, 'Unsupported AVB payload hash algorithm: '+name)
    ensure(descriptor.flags == 0, 'Unsupported AVB descriptor flags: '+name)
    embedded = getattr(descriptor, field)
    ensure(isinstance(embedded, bytes) and len(embedded) == hashlib.new(descriptor.hash_algorithm).digest_size,
           'Missing or malformed embedded AVB '+field+': '+name)


def verified_boot():
    details={n:parse_avb(n) for n in ['vbmeta','boot','dtbo',*PARTITIONS]}
    ensure(details['vbmeta']['flags']==3,'Top-level vbmeta must use development flags 3')
    ensure(details['vbmeta']['algorithm']=='SHA256_RSA4096','Unexpected top-level vbmeta algorithm')
    main=meta['vbmeta'][2]
    partition_desc=[d for d in main if isinstance(d,(avb.AvbHashDescriptor,avb.AvbHashtreeDescriptor))]
    ensure({d.partition_name for d in partition_desc}==set(['boot','dtbo',*PARTITIONS]) and len(partition_desc)==6,'Top-level descriptors do not cover exact image set')
    ensure(not any(isinstance(d,avb.AvbChainPartitionDescriptor) for d in main),'Unexpected chained vbmeta layout')
    for name in ['boot','dtbo',*PARTITIONS]:
        desc=[d for d in meta[name][2] if isinstance(d,(avb.AvbHashDescriptor,avb.AvbHashtreeDescriptor))]
        ensure(len(desc)==1 and desc[0].partition_name==name,'Unexpected partition footer descriptor: '+name)
        ensure(next(d for d in partition_desc if d.partition_name==name).encode()==desc[0].encode(),'Top-level/footer descriptor mismatch: '+name)
        ensure(details[name]['flags']==0,'Unexpected non-top-level vbmeta flags')
        d=desc[0]
        require_embedded_descriptor(name,d)
        ensure(meta[name][0] is not None,'Missing partition AVB footer: '+name)
        ensure(d.image_size==meta[name][0].original_image_size,'Footer/hash extent mismatch: '+name)
        if name in ('boot','dtbo'):ensure(details[name]['algorithm']=='NONE','Unexpected boot/DTBO footer signing algorithm')
        if name in PARTITIONS:
            ensure(isinstance(d,avb.AvbHashtreeDescriptor),'Missing filesystem hashtree descriptor')
            ensure(d.image_size==meta[name][0].original_image_size, 'Filesystem descriptor and footer disagree on original data size')
            ensure(d.image_size<=d.tree_offset and d.tree_offset+d.tree_size<=meta[name][0].vbmeta_offset,'Hashtree extent overlaps data or vbmeta')
            if d.fec_size:ensure(d.fec_offset>=d.tree_offset+d.tree_size and d.fec_offset+d.fec_size<=meta[name][0].vbmeta_offset,'FEC extent overlaps tree or vbmeta')
    public_key=A/'external/avb/test/data/testkey_rsa4096_pub.pem'
    run([sys.executable,AVB,'verify_image','--image',P/'vbmeta.img','--key',public_key],'avb-verify-all')
    run([sys.executable,AVB,'info_image','--image',P/'vbmeta.img'],'vbmeta-info')
    details['public_development_key']=info(public_key)
    details['descriptor_payloads_verified']=True
    details['hashtrees_verified']=True
    details['fec_parity_verified']=False
    details['trusted_production_chain']=False
    details['correspondence_scope']='All current top-level and partition descriptors verified together; no old build hashes carried forward'
    return details
check('full vbmeta signature, descriptors and image correspondence',verified_boot)


def textname(raw):
    text=raw.split(b'\0',1)[0].decode('ascii');ensure(re.fullmatch(r'[A-Za-z0-9_]+',text),'Invalid metadata name');return text

def metadata(blob):
    magic,major,minor,hsize=struct.unpack_from('<IHHI',blob)
    ensure(magic==0x414c5030 and major==10 and 0<=minor<=2 and hsize in (128,256),'Unexpected LP header')
    header=bytearray(blob[:hsize]);wanted=bytes(header[12:44]);header[12:44]=bytes(32)
    ensure(hashlib.sha256(header).digest()==wanted,'LP header checksum mismatch')
    tsize=struct.unpack_from('<I',blob,44)[0];ensure(hsize+tsize<=len(blob),'LP table bounds')
    table=blob[hsize:hsize+tsize];ensure(hashlib.sha256(table).digest()==blob[48:80],'LP table checksum mismatch')
    flags=struct.unpack_from('<I',blob,128)[0] if hsize>=256 else 0
    ensure(flags==0,'Unexpected LP flags, including virtual A/B')
    def rows(offset,size,fmt):
        start,count,stride=struct.unpack_from('<III',blob,offset)
        ensure(stride==size and start+count*stride<=len(table),'LP entry table bounds/size')
        return [struct.unpack_from(fmt,table,start+i*stride) for i in range(count)]
    ps=rows(80,52,'<36sIIII');es=rows(92,24,'<QIQI');gs=rows(104,48,'<36sIQ');ds=rows(116,64,'<QIIQ36sI')
    groups=[{'name':textname(n),'flags':f,'maximum_size':s} for n,f,s in gs]
    devices=[{'first_logical_sector':first,'alignment':al,'alignment_offset':ao,'size':size,'name':textname(n),'flags':f} for first,al,ao,size,n,f in ds]
    extents=[{'num_sectors':n,'target_type':t,'target_data':d,'target_source':s} for n,t,d,s in es]
    partitions=[]
    for n,attr,first,count,group in ps:
        ensure(group<len(groups) and first+count<=len(es),'LP partition table indexes')
        parts=extents[first:first+count]
        partitions.append({'name':textname(n),'attributes':attr,'group':groups[group]['name'],
            'first_extent_index':first,'extent_count':count,'size':sum(e['num_sectors']*512 for e in parts),'extents':parts})
    ensure(len({p['name'] for p in partitions})==len(partitions),'Duplicate LP partition')
    return {'version':f'{major}.{minor}','header_size':hsize,'flags':flags,'groups':groups,'block_devices':devices,
            'partitions':partitions,'metadata_sha256':hashlib.sha256(blob[:hsize+tsize]).hexdigest()}

def super_image():
    image=images['super'];primary=readat(image,4096,4096);backup=readat(image,8192,4096)
    ensure(primary==backup,'Super geometry copies differ')
    magic,size=struct.unpack_from('<II',primary)
    ensure(magic==0x616c4467 and size==52,'Unexpected LP geometry')
    g=bytearray(primary[:size]);wanted=bytes(g[8:40]);g[8:40]=bytes(32)
    ensure(hashlib.sha256(g).digest()==wanted,'LP geometry checksum mismatch')
    maximum,slots,block=struct.unpack_from('<III',primary,40)
    ensure((maximum,slots,block)==(65536,3,4096),'Wrong metadata reservation/slot count/block size')
    copies=[]
    for index in range(slots):
        first=readat(image,12288+maximum*index,maximum)
        second=readat(image,12288+maximum*slots+maximum*index,maximum)
        ensure(first==second,'Primary/backup LP metadata differ for slot '+str(index))
        copies.append(metadata(first))
    ensure(all(m==copies[0] for m in copies),'Factory metadata slots differ')
    m=copies[0];groups={x['name']:x for x in m['groups']}
    ensure(set(groups)=={'default','r1_dynamic_partitions_a','r1_dynamic_partitions_b'},'Wrong LP groups')
    for slot in 'ab':ensure(groups['r1_dynamic_partitions_'+slot]=={'name':'r1_dynamic_partitions_'+slot,'flags':0,'maximum_size':GROUP_SIZE},'Wrong group limit/flags')
    ensure(len(m['block_devices'])==1,'Expected one physical super device')
    dev=m['block_devices'][0]
    ensure(dev['name']=='super' and dev['size']==SUPER_SIZE and dev['flags']==0,'Wrong LP physical device')
    ensure(dev['first_logical_sector']*512>=12288+2*maximum*slots,'Logical data overlaps metadata reservation')
    parts={x['name']:x for x in m['partitions']}
    ensure(set(parts)=={p+'_'+s for p in PARTITIONS for s in 'ab'},'Wrong LP partition set')
    ranges=[];hashes={}
    for name,p in parts.items():
        slot=name[-1];base=name[:-2]
        ensure(p['attributes']==0 and p['group']=='r1_dynamic_partitions_'+slot,'Wrong partition attributes/group')
        if slot=='b':ensure(p['size']==0 and p['extent_count']==0,'Factory B partition unexpectedly populated');continue
        ensure(p['size']==images[base].image_size,'LP size differs from complete image: '+name)
        h=hashlib.sha256()
        for e in p['extents']:
            ensure(e['target_type']==0 and e['target_source']==0 and e['num_sectors']>0,'Unexpected non-linear LP extent')
            start=e['target_data']*512;length=e['num_sectors']*512
            ensure(start>=dev['first_logical_sector']*512 and start+length<=SUPER_SIZE,'LP extent outside usable device')
            ensure(start%block==0 and length%block==0,'LP extent not logical-block aligned')
            ranges.append((start,start+length,name))
            for off in range(0,length,1024*1024):h.update(readat(image,start+off,min(1024*1024,length-off)))
        hashes[name]=h.hexdigest();ensure(hashes[name]==sha(P/(base+'.img')),'Super payload differs from source image: '+name)
    ranges.sort()
    for left,right in zip(ranges,ranges[1:]):ensure(left[1]<=right[0],'Overlapping logical extents')
    total=sum(x['size'] for x in parts.values());ensure(total<=GROUP_SIZE,'A payload exceeds group')
    return {'geometry':{'metadata_max_size':maximum,'metadata_slot_count':slots,'logical_block_size':block,
            'sha256':hashlib.sha256(primary[:size]).hexdigest()},'metadata':m,
            'primary_backup_and_all_slots_identical':True,'logical_extent_image_sha256':hashes,
            'nonoverlapping_extents':ranges,'populated_slot':'a','empty_slot':'b',
            'group_used_bytes':total,'group_remaining_bytes':GROUP_SIZE-total,
            'raw_super_expanded_or_written':False}
check('sparse super metadata, A/B layout and actual extent payloads',super_image)

def image_directory(part,path,tag):
    text=run([DEBUGFS,'-R','ls -l -p '+path,P/(part+'.img')],tag)
    if 'File not found by ext2_lookup' in text:
        ensure(not any(line.startswith('/') for line in text.splitlines()),'Ambiguous missing image directory')
        return []
    rows=[];parsed=False
    for line in text.splitlines():
        fields=line.split('/')
        if len(fields)!=8:continue
        inode,mode,uid,gid,name,size=fields[1:7]
        if not inode.isdigit() or not mode:continue
        parsed=True
        if name in ('.','..'):continue
        ensure(re.fullmatch(r'[A-Za-z0-9_.+@-]+',name),'Unsafe image directory entry')
        rows.append({'name':name,'mode':int(mode,8),'bytes':size})
    ensure(parsed,'Unrecognized debugfs directory listing: '+part+':'+path)
    return rows

def dt_table(data):
    ensure(len(data)>=64,'Truncated DT table')
    magic,total,header,stride,count,start,page,version=struct.unpack_from('>8I',data)
    ensure((magic,header,stride,count,start,page,version)==(0xd7b7ab1e,32,32,1,32,2048,0),'Unexpected DT table geometry')
    ensure(64<=total<=len(data),'DT table outside image')
    size,offset,dtid,revision,*custom=struct.unpack_from('>8I',data,start)
    ensure(offset>=64 and offset+size<=total and dtid==revision==0 and custom==[0]*4,'Invalid DT entry')
    blob=data[offset:offset+size];ensure(blob[:4]==b'\xd0\x0d\xfe\xed','DT entry is not FDT')
    return {'total_size':total,'dtb_size':size,'offset':offset,'sha256':hashlib.sha256(data[:total]).hexdigest()},blob

def pstore_actual():
    symbol='__android_log_pmsg_file_read_from_backend';out={}
    items=[('system-liblog',extracted[('system','/system/lib64/liblog.so')],True,False),
           ('recovery-liblog',recovery_liblog,True,False),
           ('recovery-persist',extracted[('system','/system/bin/recovery-persist')],False,True),
           ('dumpstate',extracted[('system','/system/bin/dumpstate')],False,False)]
    for name,path,provider,consumer in items:
        item=elf(path,name);text=run([READELF,'--wide','--dyn-syms',path],name+'-symbols')
        lines=[line.strip() for line in text.splitlines() if symbol in line]
        if provider:ensure(any(symbol+'@@LIBLOG_PRIVATE' in line and ' UND ' not in line for line in lines),'Missing private liblog export')
        if consumer:ensure('liblog.so' in item['needed'] and any(symbol+'@LIBLOG_PRIVATE' in line and ' UND ' in line for line in lines),'Missing private pmsg import')
        paths=[b'/sys/fs/pstore/pmsg-ramoops-0',b'/sys/fs/pstore/pmsg-pstore_blk-0'] if provider else [b'/sys/fs/pstore/console-ramoops',b'/sys/fs/pstore/console-pstore_blk-0']
        if consumer:paths += [b'/sys/fs/pstore/pmsg-ramoops-0',b'/sys/fs/pstore/pmsg-pstore_blk-0']
        data=path.read_bytes()
        for token in paths:ensure(token+b'\0' in data,'Missing current pstore backend string '+name)
        item['private_api_symbols']=lines;item['backend_paths']=[x.decode() for x in paths];out[name]=item
    for name,source in [('recovery-persist','bootable/recovery/recovery-persist.rc'),('dumpstate','frameworks/native/cmds/dumpstate/dumpstate.rc')]:
        path=extracted[('system','/system/etc/init/'+name+'.rc')]
        ensure(path.read_bytes()==(A/source).read_bytes(),'Packaged pstore collector init differs')
    return {'actual_packaged_artifacts':out,'old_build_hashes_reused':False,'runtime_persistence_verified':False}
check('current packaged pstore collector and liblog implementations',pstore_actual)

def allocator_apex():
    apex_path = globals()['extracted'][('vendor','/apex/com.android.hardware.graphics.allocator.minigbm_upstream.apex')]
    install = info(apex_path)
    payload = H/'apex_payload.img'
    with zipfile.ZipFile(apex_path) as archive:
        ensure(archive.namelist().count('apex_payload.img') == 1, 'Missing/duplicate APEX payload')
        payload.write_bytes(archive.read('apex_payload.img'))
        archive_names = archive.namelist()
        zip_manifest = archive.read('apex_manifest.pb')
    with payload.open('rb') as stream:
        stream.seek(1024)
        superblock = stream.read(58)
    filesystem = 'erofs' if superblock[:4] == b'\xe2\xe1\xf5\xe0' else 'ext4' if superblock[56:58] == b'\x53\xef' else None
    ensure(filesystem is not None, 'Unsupported APEX payload filesystem')
    extracted, listing = {}, []
    def walk(directory):
        text = command([DEBUGFS,'-R','ls -l -p '+directory,payload])
        for line in text.splitlines():
            fields = line.split('/')
            if len(fields) != 8: continue
            inode,mode,uid,gid,name,size = fields[1:7]
            if not name or name in ('.','..'): continue
            ensure('/' not in name and name not in ('.','..') and re.fullmatch(r'[A-Za-z0-9_.+@-]+',name), 'Unexpected APEX filename')
            path = directory.rstrip('/')+'/'+name
            mode = int(mode,8)
            listing.append({'path':path,'mode':oct(mode),'size':size})
            if stat.S_ISDIR(mode): walk(path)
            elif stat.S_ISREG(mode):
                destination = H/'apex-files'/path.lstrip('/')
                destination.parent.mkdir(parents=True,exist_ok=True)
                command([DEBUGFS,'-R',f'dump {path} {destination}',payload])
                ensure(destination.is_file() and destination.stat().st_size == int(size), 'APEX dump failed '+path)
                extracted[path] = destination
    if filesystem == 'ext4':
        walk('/')
    else:
        # Same host read-only extraction used by AOSP deapexer for EROFS.
        fsck = O/'host/linux-x86/bin/fsck.erofs'
        destination = Path(tempfile.mkdtemp(prefix='apex-erofs-',dir=H))
        command([fsck,'--extract='+str(destination),'--no-preserve-owner','--no-preserve-perms',payload], 'apex-erofs-extract')
        r['erofs_host_tool'] = info(fsck)
        for parent,dirs,files in os.walk(destination,followlinks=False):
            for name in dirs+files:
                path = Path(parent)/name
                relative = '/'+str(path.relative_to(destination))
                metadata = path.lstat()
                ensure(stat.S_ISREG(metadata.st_mode) or stat.S_ISDIR(metadata.st_mode) or stat.S_ISLNK(metadata.st_mode), 'Unexpected non-file in built APEX '+relative)
                listing.append({'path':relative,'mode':oct(metadata.st_mode),'size':metadata.st_size})
                if stat.S_ISREG(metadata.st_mode):
                    ensure(path.resolve().is_relative_to(destination), 'Extracted regular file escapes review directory')
                    extracted[relative] = path
    (H/'apex-listing.json').write_text(json.dumps(listing,indent=2)+'\n')
    ensure(zip_manifest == read(extracted['/apex_manifest.pb']), 'ZIP/payload protobuf manifests differ')
    deapexer = O/'host/linux-x86/bin/deapexer'
    manifest = json.loads(command([deapexer,'info',apex_path], 'apex-manifest'))
    ensure(manifest['name'] == 'com.android.hardware.graphics.allocator' and int(manifest['version']) == 1, 'Wrong APEX manifest identity')
    ensure(manifest.get('vendorBootstrap') is True, 'Missing vendor bootstrap manifest flag')
    android_manifest = command([O/'host/linux-x86/bin/aapt2','dump','xmltree',apex_path,'--file','AndroidManifest.xml'], 'apex-android-manifest')
    ensure('package="com.android.hardware.graphics.allocator"' in android_manifest, 'Wrong APEX Android package')
    r['apex_manifest_host_tool'] = info(deapexer)
    required = ['/bin/hw/android.hardware.graphics.allocator-service.minigbm_upstream','/lib64/hw/mapper.minigbm_upstream.so','/lib64/libminigbm_gralloc_upstream.so','/etc/vintf/allocator.minigbm_upstream.xml','/etc/vintf/mapper.minigbm_upstream.xml','/etc/linker.config.pb','/etc/com.android.hardware.graphics.allocator.minigbm_upstream.rc']
    for name in required: ensure(name in extracted, 'APEX missing '+name)
    rc = read(extracted['/etc/com.android.hardware.graphics.allocator.minigbm_upstream.rc'])
    ensure(b'/apex/com.android.hardware.graphics.allocator/bin/hw/android.hardware.graphics.allocator-service.minigbm_upstream' in rc, 'Wrong APEX service path')
    binaries = {}
    for name,path in extracted.items():
        with path.open('rb') as stream: magic = stream.read(4)
        if magic == b'\x7fELF': binaries[name] = elf(path,'apex-'+path.name)
    ensure(len(binaries) >= 3, 'APEX ELF payload incomplete')
    packaged_names = {Path(name).name for name in binaries}
    external_needed = sorted({name for binary in binaries.values() for name in binary['needed']} - packaged_names)
    ensure(set(external_needed) <= set(manifest.get('requireNativeLibs',[])), 'APEX ELF imports missing from declared external native libraries')
    # assemble_vintf emits schema 9, canonical fqnames and an input comment.
    expected_vintf = {
        'allocator': '<manifest version="9.0" type="device"><hal format="aidl"><name>android.hardware.graphics.allocator</name><version>3</version><fqname>IAllocator/default</fqname></hal></manifest>',
        'mapper': '<manifest version="9.0" type="device"><hal format="native"><name>mapper</name><fqname>@5.0/minigbm_upstream</fqname></hal></manifest>',
    }
    def semantic(element):
        return [element.tag,dict(element.attrib),(element.text or '').strip(),[semantic(child) for child in element]]
    vintf = {}
    for name,expected in expected_vintf.items():
        actual = semantic(ET.fromstring(read(extracted['/etc/vintf/'+name+'.minigbm_upstream.xml'])))
        ensure(actual == semantic(ET.fromstring(expected)), 'Unexpected packaged VINTF declaration '+name)
        vintf[name] = actual
    ensure(not any(name.startswith('/lib/') for name in binaries), 'Unexpected 32-bit APEX variant')
    return {**install,'manifest':manifest,'payload':info(payload),'payload_filesystem':filesystem,'archive_members':archive_names,
            'required_files':{name:info(extracted[name]) for name in required},'elfs':binaries,'external_needed_declared_in_manifest':external_needed,'vintf_declarations':vintf,
            'signature_trust_verified':False,'runtime_dependency_resolution_verified':False}
check('current allocator APEX payload, ARM64 libraries, init and VINTF',allocator_apex)
guard.watch([O/'host/linux-x86/bin/aapt2',O/'host/linux-x86/bin/deapexer'])
